% ============================================
%         FRACTIONS
% ============================================
\newcommand\der{\mathrm d}
\newcommand*{\ddt}{
    \frac{\der}{\der t}
}
\newcommand*{\ddtt}{
    \tfrac{\der}{\der t}
}
\newcommand*{\ddfrac}[2]{
    \frac{\der {#1}}{\der {#2}}
}
\newcommand*{\ddtfrac}[2]{
    \tfrac{\der {#1}}{\der {#2}}
}
\newcommand*{\ppfrac}[2]{
    \frac{\partial {#1}}{\partial {#2}}
}
\newcommand*{\pptfrac}[2]{
    \tfrac{\partial {#1}}{\partial {#2}}
}
